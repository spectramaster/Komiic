```cpp
// core/GameEngine.cpp
void GameEngine::init() {
    m_levelManager = new LevelManager(this);
    m_achievementSystem = new AchievementSystem(this);
    m_emulator = new STM32Emulator(this);
    m_compiler = new CodeCompiler(this);

    // 连接信号
    connect(m_compiler, &CodeCompiler::compileSuccess,
            this, &GameEngine::onCompileSuccess);
    connect(m_compiler, &CodeCompiler::compileError,
            this, &GameEngine::onCompileError);

    // 加载用户数据
    UserProfile::instance().load();
}

void GameEngine::submitCode(const QString& code) {
    // 1. 编译代码
    emit statusChanged("正在编译...");
    m_compiler->compile(code);
}

void GameEngine::onCompileSuccess(const QString& binaryPath) {
    // 2. 加载到模拟器
    emit statusChanged("正在运行...");
    m_emulator->loadAndRun(binaryPath);

    // 3. 检查任务目标
    QTimer::singleShot(100, this, [this]() {
        if(checkMissionComplete()) {
            handleLevelComplete();
        }
    });
}

bool GameEngine::checkMissionComplete() {
    Level* currentLevel = m_levelManager->currentLevel();
    if(!currentLevel) return false;

    // 检查所有任务目标
    for(const auto& objective : currentLevel->objectives()) {
        if(!objective.isCompleted(m_emulator)) {
            return false;
        }
    }
    return true;
}
```

```cpp
// core/LevelManager.cpp
Level* LevelManager::loadLevel(int chapterId, int levelId) {
    QString path = QString(":/resources/levels/chapter%1/%2-%3.json")
                       .arg(chapterId).arg(chapterId).arg(levelId);

    QFile file(path);
    if(!file.open(QIODevice::ReadOnly)) {
        qWarning() << "无法打开关卡文件:" << path;
        return nullptr;
    }

    QJsonDocument doc = QJsonDocument::fromJson(file.readAll());
    QJsonObject json = doc.object();

    Level* level = new Level;
    level->chapterId = chapterId;
    level->levelId = levelId;
    level->title = json["title"].toString();
    level->description = json["description"].toString();
    level->storyText = json["story"].toString();
    level->templateCode = json["template"].toString();
    level->xpReward = json["xp_reward"].toInt();
    level->crystalReward = json["crystal_reward"].toInt();

    // 解析任务目标
    QJsonArray objectives = json["objectives"].toArray();
    for(const QJsonValue& obj : objectives) {
        Objective objective;
        objective.type = stringToObjectiveType(
            obj["type"].toString()
        );
        objective.parameters = obj["parameters"].toObject().toVariantMap();
        level->objectives.append(objective);
    }

    // 解析星级标准
    QJsonObject stars = json["stars"].toObject();
    level->threeStar.lines = stars["3star"]["lines"].toInt();
    level->threeStar.memory = stars["3star"]["memory"].toInt();
    level->threeStar.time = stars["3star"]["time"].toInt();

    m_currentLevel = level;
    return level;
}
```

```cpp
// emulator/GPIO.cpp
void GPIO::writePin(int pin, bool state) {
    if(pin < 0 || pin > 15) return;

    // 更新ODR寄存器
    if(state) {
        ODR |= (1 << pin);
    } else {
        ODR &= ~(1 << pin);
    }

    // 更新引脚状态
    if(m_pinStates[pin] != state) {
        m_pinStates[pin] = state;
        emit pinChanged(pin, state);
    }
}
```

```cpp
// emulator/Timer.cpp
class Timer : public QObject {
    Q_OBJECT
public:
    // 寄存器
    uint32_t CR1;   // 控制寄存器1
    uint32_t CNT;   // 计数器
    uint32_t ARR;   // 自动重装载寄存器
    uint32_t PSC;   // 预分频器

    void start();
    void stop();
    void tick();  // 系统时钟调用

signals:
    void overflow();
    void compare(int channel);

private:
    QTimer* m_timer;
    uint32_t m_prescalerCounter;
};

void Timer::tick() {
    if(!(CR1 & 0x01)) return;  // 未使能

    m_prescalerCounter++;
    if(m_prescalerCounter >= PSC) {
        m_prescalerCounter = 0;
        CNT++;

        if(CNT >= ARR) {
            CNT = 0;
            emit overflow();
        }
    }
}
```

```cpp
// emulator/STM32Emulator.cpp
STM32Emulator::STM32Emulator(QObject* parent)
    : QObject(parent)
{
    // 初始化GPIO (A-E)
    for(char port = 'A'; port <= 'E'; port++) {
        GPIO* gpio = new GPIO(port, this);
        m_gpios[port] = gpio;

        // 连接信号
        connect(gpio, &GPIO::pinChanged,
                this, [this, port](int pin, bool state) {
            emit gpioChanged(port, pin, state);
        });
    }

    // 初始化定时器
    for(int i = 1; i <= 4; i++) {
        Timer* timer = new Timer(i, this);
        m_timers[i] = timer;

        connect(timer, &Timer::overflow,
                this, [this, i]() {
            emit timerTick(i);
        });
    }

    // 系统时钟 (简化为1kHz)
    m_systemClock = new QTimer(this);
    connect(m_systemClock, &QTimer::timeout,
            this, &STM32Emulator::clockTick);
    m_systemClock->setInterval(1);  // 1ms
}
```

### 1. 增量编译

```cpp
class IncrementalCompiler {
public:
    void compile(const QString& code) {
        // 计算代码哈希
        QString hash = QCryptographicHash::hash(
            code.toUtf8(),
            QCryptographicHash::Md5
        ).toHex();

        // 检查缓存
        if(m_cache.contains(hash)) {
            emit compileSuccess(m_cache[hash]);
            return;
        }

        // 全量编译
        doFullCompile(code);
    }

private:
    QMap<QString, QString> m_cache;  // 哈希 -> 二进制路径
};
```

```cpp
        // 在单独线程运行模拟器
        QThread* emulatorThread = new QThread;
        m_core->moveToThread(emulatorThread);

        connect(emulatorThread, &QThread::started,
                m_core, &STM32Core::run);

        emulatorThread->start();
    }
};
```
